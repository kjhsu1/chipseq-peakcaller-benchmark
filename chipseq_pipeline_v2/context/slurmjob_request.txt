#!/bin/bash
#SBATCH --job-name=real_peaks128
#SBATCH --account=ikorfgrp
#SBATCH --partition=high
#SBATCH --output=/quobyte/ikorfgrp/home/kjhsu/results/chipseq_pipeline_v2/slurm_jobs/%j/%x.out
#SBATCH --error=/quobyte/ikorfgrp/home/kjhsu/results/chipseq_pipeline_v2/slurm_jobs/%j/%x.err
#SBATCH --cpus-per-task=4
#SBATCH --mem=4G
#SBATCH --time=3:00:00

set -euo pipefail

source /etc/profile
module load conda

ENV_NAME=chipseq_align
PIPE=/home/kjhsu/Code/Korflab/chipseq-peakcaller-benchmark/chipseq_pipeline_v2
DATA=/quobyte/ikorfgrp/home/kjhsu/results/chipseq_pipeline_v2

mkdir -p "$DATA/slurm_jobs/${SLURM_JOB_ID}"
mkdir -p "$DATA/results" "$DATA/archived_results/sweep_logs"

cd "$PIPE"

rm -rf results archived_results
ln -sfn "$DATA/results" results
ln -sfn "$DATA/archived_results" archived_results

rm -rf "$DATA/results"/*
ts=$(date +%Y%m%d_%H%M%S)
log="$DATA/archived_results/sweep_logs/realistic_peaks_integrated_enrich128_${ts}.log"

echo "[$(date)] START configs/realistic_peaks_integrated_enrich128.yaml" | tee "$log"

conda run -n "$ENV_NAME" snakemake -j 4 -s Snakefile.py \
  --configfile configs/realistic_peaks_integrated_enrich128.yaml \
  --latency-wait=60 2>&1 | tee -a "$log"

dest="$DATA/archived_results/realistic_peaks_integrated_enrich128_${ts}"
cp -a "$DATA/results" "$dest"
echo "archived -> $dest" | tee -a "$log"

rm -rf "$DATA/results"/*
echo "[$(date)] END configs/realistic_peaks_integrated_enrich128.yaml" | tee -a "$log"
